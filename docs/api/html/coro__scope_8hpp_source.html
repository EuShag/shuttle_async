68 // Жесткий release: публикуем все данные корутины для ждущего потока

75 void waitForSignal() noexcept {
76 // Защита от ложных пробуждений (Spurious Wakeups)
77 // Если проснулись случайно, а флаг все еще false — уходим спать дальше
78 while (!m_ready.load(std::memory_order_acquire)) {
79 // Передаем false: засыпаем ТОЛЬКО если m_ready равен false.
80 // Если он уже true, wait() мгновенно вернет управление без сна.
81 m_ready.wait(false, std::memory_order_acquire);
82 }
83
84 // Сбрасываем в false для следующих итераций (например, для joinTask в деструкторе)
85 m_ready.store(false, std::memory_order_relaxed);
86 }

223 template<typename F, typename... Args>
224 requires CoroTaskFactory<F, Args...>
225 auto launch(F&& func, LaunchOptions launchOptions, Args&&... args) noexcept {
226 auto coro = std::invoke(
227 std::forward<F>(func),
228 this->createChild(launchOptions),
229 std::forward<Args>(args)...
230 );
231 if (!coro.has_value()) [[unlikely]] {
232 return coro;
233 }
234
235 if (launchOptions.mode == LaunchMode::Async) {
236 auto childHandle = coro.get_handle();
237 if (launchOptions.explicitExecutor == nullptr) [[unlikely]] {
238 childHandle.destroy();
239 return decltype(coro){Error::executor_unavailable};
240 }
241
242 auto queueResult = coro.start(launchOptions.explicitExecutor);
243 if (!queueResult) [[unlikely]] {
244 childHandle.destroy();
245 return decltype(coro){queueResult.error()};
246 }
247 }
248
249 return coro;
250 }
