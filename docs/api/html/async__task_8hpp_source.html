711 decltype(auto) get() {
712 m_task.requireValid();
713 m_task.requireSingleConsumer();
714 m_task.startForAwait();
715 auto threadContextGuard = m_executor.initializeThreadContext();
716
717 // 1. Obtain the awaitable object by passing the address of our internal m_executor.
718 // This directly accesses the internal state of the AsyncTask.
719 auto awaitable = m_task.m_future.prepareAwaitable(&m_executor);
720 auto* state = awaitable.m_state;
721
722 // -----------------------------------------------------------------
723 // FAST-PATH STAGE 1: Data is ALREADY available (Producer finished before get)
724 // -----------------------------------------------------------------
725 if (state->isSignaled()) {
726 if constexpr (std::same_as<T, void>) return;
727 else return state->extractValue();
728 }
729
730 // -----------------------------------------------------------------
731 // FAST-PATH STAGE 2: Data race with the producer during registration.
732 // If tryRegisterWaiter returns false, it implies that while we were
733 // writing pointers, the producer IMMEDIATELY transitioned the state to Signaled!
734 // -----------------------------------------------------------------
735 if (!state->tryRegisterWaiter(std::noop_coroutine(), &m_executor)) {
736 if constexpr (std::same_as<T, void>) return;
737 else return state->extractValue();
738 }
739
740 // -----------------------------------------------------------------
741 // SLOW-PATH STAGE 3: Data is absent, waiter is successfully registered.
742 // Force the physical OS thread into deep kernel-level suspension via futex/WaitOnAddress.
743 // Protection against spurious wake-ups (while loop) is already handled inside waitForSignal!
744 // -----------------------------------------------------------------
745 m_executor.waitForSignal();
746
747 // The thread has woken up; data is guaranteed to be available.
748 // Extract the r-value reference without performing a single copy operation!
749 if constexpr (std::same_as<T, void>) return;
750 else return state->extractValue();
751 }
