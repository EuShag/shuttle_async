118 void waitForSignal() noexcept {
119 // Защита от ложных пробуждений (Spurious Wakeups)
120 // Если проснулись случайно, а флаг все еще false — уходим спать дальше
121 while (!m_ready.load(std::memory_order_acquire)) {
122 // Передаем false: засыпаем ТОЛЬКО если m_ready равен false.
123 // Если он уже true, wait() мгновенно вернет управление без сна.
124 m_ready.wait(false, std::memory_order_acquire);
125 }
126
127 // Сбрасываем в false для следующих итераций (например, для joinTask в деструкторе)
128 m_ready.store(false, std::memory_order_relaxed);
129 }

409 template<typename F, typename... Args>
410 requires CoroTaskFactory<F, Args...>
411 auto launch(F&& func, LaunchOptions launchOptions, Args&&... args) noexcept {
412 auto coro = std::invoke(
413 std::forward<F>(func),
414 this->createChild(launchOptions),
415 std::forward<Args>(args)...
416 );
417 if (!coro.has_value()) [[unlikely]] {
418 return coro;
419 }
420
421 if (launchOptions.mode == LaunchMode::Async) {
422 auto childHandle = coro.get_handle();
423 if (launchOptions.explicitExecutor == nullptr) [[unlikely]] {
424 childHandle.destroy();
425 return decltype(coro){Error::executor_unavailable};
426 }
427
428 auto queueResult = coro.start(launchOptions.explicitExecutor);
429 if (!queueResult) [[unlikely]] {
430 childHandle.destroy();
431 return decltype(coro){queueResult.error()};
432 }
433 }
434
435 return coro;
436 }
