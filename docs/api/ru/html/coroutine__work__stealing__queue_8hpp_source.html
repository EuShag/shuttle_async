120 const int64_t bottom = _bottom.load(std::memory_order_relaxed);
121 // Тор читается с relaxed, так как его актуальность критична только при переполнении,

148 std::coroutine_handle<> pop() noexcept {
149 int64_t bottom = _bottom.load(std::memory_order_relaxed) - 1;
150 _bottom.store(bottom, std::memory_order_relaxed);
151
152 // КРИТИЧЕСКИЙ БАРЬЕР: Предотвращает переупорядочивание Store(_bottom) и Load(_top) процессором
153 std::atomic_thread_fence(std::memory_order_seq_cst);

155 int64_t top = _top.load(std::memory_order_relaxed);
156
157 if (top <= bottom) {

159 if (top == bottom) {

163 std::memory_order_seq_cst,
164 std::memory_order_relaxed
165 )) {

167 }
168 _bottom.store(bottom + 1, std::memory_order_relaxed);
169 }

171 }

173 _bottom.store(bottom + 1, std::memory_order_relaxed);
174 return {};
